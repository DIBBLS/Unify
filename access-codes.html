// Run once after generating a batch — writes codes to Firestore
// Use this in your admin console or a one-off Node script

async function writeCodeBatch(codes) {
  var batch = db.batch();

  codes.forEach(function(code) {
    var ref = db.collection("accessCodes").doc(code.code);
    batch.set(ref, {
      tier:      code.tier,
      isUsed:    false,
      createdAt: firebase.firestore.FieldValue.serverTimestamp(),
      expiresAt: firebase.firestore.Timestamp.fromDate(
                   new Date(code.expiresAt)
                 ),
      usedBy:    null
    });
  });

  await batch.commit();
  console.log("Written:", codes.length, "codes");
}

async function redeemCode(code, userId) {
  var codeRef = db.collection("accessCodes")
                  .doc(code.trim().toUpperCase());
  var userRef = db.collection("users").doc(userId);

  try {
    await db.runTransaction(async function(txn) {
      var snap = await txn.get(codeRef);

      if (!snap.exists)
        throw new Error("Code not found. Check and try again.");

      var data = snap.data();

      if (data.isUsed)
        throw new Error("This code has already been used.");

      if (data.expiresAt.toDate() < new Date())
        throw new Error("This code has expired.");

      // Lock the code
      txn.update(codeRef, {
        isUsed: true,
        usedBy: userId,
        usedAt: firebase.firestore.FieldValue.serverTimestamp()
      });

      // Activate subscription
      txn.set(userRef, {
        subscription: {
          active:      true,
          tier:        data.tier,
          activatedAt: firebase.firestore.FieldValue.serverTimestamp(),
          expiresAt:   data.expiresAt
        }
      }, { merge: true });
    });

    return { success: true,
             message: "Access activated successfully." };

  } catch (err) {
    return { success: false, message: err.message };
  }
}
match /accessCodes/{code} {
  allow read:  if request.auth != null;
  allow create: if isAdmin();
  // Students can only mark a code as used
  allow update: if request.auth != null
    && resource.data.isUsed == false
    && resource.data.expiresAt > request.time
    && request.resource.data.diff(resource.data)
         .affectedKeys().hasOnly(['isUsed','usedBy','usedAt'])
    && request.resource.data.isUsed == true
    && request.resource.data.usedBy == request.auth.uid;
}

match /users/{uid} {
  // Allow subscription activation via code redemption
  allow update: if request.auth != null
    && request.auth.uid == uid
    && request.resource.data.diff(resource.data)
         .affectedKeys().hasOnly(['subscription']);
}
